cmake_minimum_required(VERSION 3.16)
project(lite3_bridge_compatibility_review)
set(CMAKE_CXX_STANDARD 14)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(transfer_interfaces REQUIRED)
find_package(Threads REQUIRED)
get_filename_component(REVIEW_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(VENDOR "${REVIEW_ROOT}/vendor/Lite3_ROS/src/transfer/src")
# Compile/link checks only: no install targets, launch commands or executable tests.
foreach(pair "jetson2motion_review:Jetson2Motion.cpp" "jetson2app_review:Jetson2App.cpp" "sensor_checker_review:SensorChecker.cpp")
  string(REPLACE ":" ";" parts "${pair}")
  list(GET parts 0 target)
  list(GET parts 1 source)
  add_executable(${target} "${VENDOR}/${source}")
  ament_target_dependencies(${target} rclcpp std_msgs sensor_msgs nav_msgs geometry_msgs
    tf2_msgs tf2 tf2_ros tf2_geometry_msgs transfer_interfaces)
  target_link_libraries(${target} Threads::Threads)
endforeach()
add_executable(jetson2motion_shutdown_review "${REVIEW_ROOT}/review/bridge_candidate/src/Jetson2Motion.cpp")
ament_target_dependencies(jetson2motion_shutdown_review rclcpp std_msgs sensor_msgs nav_msgs
  geometry_msgs tf2_msgs tf2 tf2_ros tf2_geometry_msgs transfer_interfaces)
target_link_libraries(jetson2motion_shutdown_review Threads::Threads)
