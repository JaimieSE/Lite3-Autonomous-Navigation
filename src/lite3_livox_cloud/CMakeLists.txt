cmake_minimum_required(VERSION 3.8)
project(lite3_livox_cloud)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(livox_ros_driver2 REQUIRED)
find_package(sensor_msgs REQUIRED)
add_executable(livox_cloud src/livox_cloud.cpp)
ament_target_dependencies(livox_cloud rclcpp livox_ros_driver2 sensor_msgs)
install(TARGETS livox_cloud DESTINATION lib/${PROJECT_NAME})
ament_package()
